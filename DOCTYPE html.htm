<script>

    // 6道角度計算題目資料庫 (包含繪圖專用參數)

    const angleQuestions = [

        {

            title: "(1) 求平角上的 ∠A 是幾度？ (提示：180° - 120°)",

            type: "flat", given: 120, label: "A",

            options: ["① 60 度", "② 45 度", "③ 155 度", "④ 210 度"], ans: 0

        },

        {

            title: "(2) 求直角內的 ∠B 是幾度？ (提示：90° - 45°)",

            type: "right", given: 45, label: "B",

            options: ["① 60 度", "② 45 度", "③ 155 度", "④ 210 度"], ans: 1

        },

        {

            title: "(3) 求平角上的 ∠C 是幾度？ (提示：180° - 25°)",

            type: "flat_rev", given: 25, label: "C",

            options: ["① 60 度", "② 155 度", "③ 45 度", "④ 210 度"], ans: 1

        },

        {

            title: "(4) 求優角另一側的 ∠D 是幾度？ (提示：360° - 150°)",

            type: "reflex", given: 150, label: "D",

            options: ["① 210 度", "② 270 度", "③ 73 度", "④ 155 度"], ans: 0

        },

        {

            title: "(5) 求直角交錯的 ∠E 是幾度？ (提示：360° - 90°)",

            type: "cross", given1: 137, given2: 133, label: "E",

            options: ["① 210 度", "② 270 度", "③ 73 度", "④ 45 度"], ans: 1

        },

        {

            title: "(6) 求合併後的 ∠F 是幾度？ (提示：53° + 20°)",

            type: "combine", given1: 53, given2: 20, label: "F",

            options: ["① 210 度", "② 73 度", "③ 270 度", "④ 155 度"], ans: 1

        }

    ];

 

    let currentIndex = 0;

    let score = 0;

    let lives = 3;

 

    // 音效系統

    let audioCtx = null;

    function initAudio() {

        if (!audioCtx) {

            audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        }

        if (audioCtx.state === 'suspended') audioCtx.resume();

    }

 

    function playCorrectSound() {

        initAudio();

        if (!audioCtx) return;

        let now = audioCtx.currentTime;

        [523.25, 659.25, 783.99, 1046.50].forEach((freq, idx) => {

            let osc = audioCtx.createOscillator();

            let gain = audioCtx.createGain();

            osc.type = 'triangle';

            osc.frequency.setValueAtTime(freq, now + idx * 0.08);

            gain.gain.setValueAtTime(0.3, now + idx * 0.08);

            gain.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.08 + 0.25);

            osc.connect(gain);

            gain.connect(audioCtx.destination);

            osc.start(now + idx * 0.08);

            osc.stop(now + idx * 0.08 + 0.25);

        });

    }

 

    function playWrongSound() {

        initAudio();

        if (!audioCtx) return;

        let now = audioCtx.currentTime;

        [220, 165].forEach((freq, idx) => {

            let osc = audioCtx.createOscillator();

            let gain = audioCtx.createGain();

            osc.type = 'sawtooth';

            osc.frequency.setValueAtTime(freq, now + idx * 0.12);

            gain.gain.setValueAtTime(0.25, now + idx * 0.12);

            gain.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.12 + 0.2);

            osc.connect(gain);

            gain.connect(audioCtx.destination);

            osc.start(now + idx * 0.12);

            osc.stop(now + idx * 0.12 + 0.2);

        });

    }

 

    // 背景音樂

    let bgmInterval = null;

    let isBgmPlaying = false;

    function toggleBGM() {

        initAudio();

        isBgmPlaying = !isBgmPlaying;

        const btn = document.getElementById("bgm-toggle");

        if (isBgmPlaying) {

            btn.innerText = "🎵 音樂: 開";

            startBGM();

        } else {

            btn.innerText = "🎵 音樂: 關";

            stopBGM();

        }

    }

 

    function startBGM() {

        if (bgmInterval) return;

        const notes = [329.63, 329.63, 392.00, 392.00, 329.63, 329.63, 261.63];

        let index = 0;

        bgmInterval = setInterval(() => {

            if (!isBgmPlaying || !audioCtx) return;

            let osc = audioCtx.createOscillator();

            let gain = audioCtx.createGain();

            osc.type = 'sine';

            osc.frequency.setValueAtTime(notes[index], audioCtx.currentTime);

            gain.gain.setValueAtTime(0.08, audioCtx.currentTime);

            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.25);

            osc.connect(gain);

            gain.connect(audioCtx.destination);

            osc.start();

            osc.stop(audioCtx.currentTime + 0.25);

            index = (index + 1) % notes.length;

        }, 300);

    }

 

    function stopBGM() {

        if (bgmInterval) {

            clearInterval(bgmInterval);

            bgmInterval = null;

        }

    }

 

    function startGame() {

        initAudio();

        currentIndex = 0;

        score = 0;

        lives = 3;

 

        document.getElementById("start-screen").classList.remove("active");

        document.getElementById("end-screen").classList.remove("active");

        document.getElementById("game-screen").classList.add("active");

 

        loadQuestion();

    }

 

    function loadQuestion() {

        if (lives <= 0) {

            endGame(false);

            return;

        }

        if (currentIndex >= angleQuestions.length) {

            endGame(true);

            return;

        }

 

        const item = angleQuestions[currentIndex];

 

        document.getElementById("current-q").innerText = currentIndex + 1;

        document.getElementById("score").innerText = score;

        document.getElementById("lives").innerText = "❤️".repeat(lives);

        document.getElementById("question-text").innerText = item.title;

        document.getElementById("feedback").innerText = "";

 

        // 繪製對應的幾何角度圖形

        drawAngle(item);

 

        const mcOptions = document.getElementById("mc-options");

        mcOptions.innerHTML = "";

        item.options.forEach((opt, idx) => {

            let btn = document.createElement("button");

            btn.className = "option-btn";

            btn.innerText = opt;

            btn.onclick = () => checkAnswer(idx, item.ans);

            mcOptions.appendChild(btn);

        });

    }

 

    // Canvas 幾何圖形繪製函數

    function drawAngle(item) {

        const canvas = document.getElementById("angleCanvas");

        const ctx = canvas.getContext("2d");

        ctx.clearRect(0, 0, canvas.width, canvas.height);

 

        ctx.strokeStyle = "#2d3436";

        ctx.lineWidth = 2.5;

        ctx.fillStyle = "#2d3436";

        ctx.font = "bold 14px 'Segoe UI'";

 

        const cx = 130, cy = 110;

 

        if (item.type === "flat") {

            // 平角 (1)

            ctx.beginPath();

            ctx.moveTo(30, cy); ctx.lineTo(230, cy); ctx.stroke();

            ctx.beginPath();

            ctx.moveTo(cx, cy); ctx.lineTo(cx - 40, cy - 60); ctx.stroke();

            // 標示角度

            ctx.fillText(`${item.given}°`, cx + 25, cy - 25);

            ctx.fillText(`∠${item.label}`, cx - 25, cy - 25);

        } else if (item.type === "right") {

            // 直角 (2)

            ctx.beginPath();

            ctx.moveTo(cx, cy + 30); ctx.lineTo(cx, cy - 50); ctx.stroke();

            ctx.moveTo(cx, cy + 30); ctx.lineTo(cx + 60, cy + 30); ctx.stroke();

            ctx.beginPath();

            ctx.moveTo(cx, cy + 30); ctx.lineTo(cx + 45, cy - 35); ctx.stroke();

            ctx.fillText(`${item.given}°`, cx + 30, cy);

            ctx.fillText(`∠${item.label}`, cx + 8, cy - 5);

        } else if (item.type === "flat_rev") {

            // 平角反向 (3)

            ctx.beginPath();

            ctx.moveTo(30, cy); ctx.lineTo(230, cy); ctx.stroke();

            ctx.beginPath();

            ctx.moveTo(cx, cy); ctx.lineTo(cx + 50, cy - 50); ctx.stroke();

            ctx.fillText(`${item.given}°`, cx - 50, cy - 20);

            ctx.fillText(`∠${item.label}`, cx + 25, cy - 20);

        } else if (item.type === "reflex") {

            // 優角 (4)

            ctx.beginPath();

            ctx.moveTo(50, cy - 30); ctx.lineTo(cx, cy); ctx.lineTo(210, cy); ctx.stroke();

            ctx.fillText(`${item.given}°`, cx + 25, cy - 20);

            ctx.fillText(`∠${item.label}`, cx - 35, cy + 25);

            // 弧線標示

            ctx.beginPath();

            ctx.arc(cx, cy, 25, 0, Math.PI * 1.2);

            ctx.stroke();

        } else if (item.type === "cross") {

            // 十字交錯 (5)

            ctx.beginPath();

            ctx.moveTo(70, cy); ctx.lineTo(190, cy); ctx.stroke();

            ctx.beginPath();

            ctx.moveTo(cx, cy - 50); ctx.lineTo(cx, cy + 50); ctx.stroke();

            // 直角記號

            ctx.strokeRect(cx - 12, cy, 12, 12);

            ctx.fillText(`${item.given1}°`, cx - 45, cy - 20);

            ctx.fillText(`${item.given2}°`, cx + 15, cy + 30);

            ctx.fillText(`∠${item.label}`, cx + 35, cy - 20);

        } else if (item.type === "combine") {

            // 併排角度 (6)

            ctx.beginPath();

            ctx.moveTo(cx, cy + 30); ctx.lineTo(cx - 60, cy - 30); ctx.stroke();

            ctx.beginPath();

            ctx.moveTo(cx, cy + 30); ctx.lineTo(cx + 10, cy - 50); ctx.stroke();

            ctx.beginPath();

            ctx.moveTo(cx, cy + 30); ctx.lineTo(cx + 60, cy - 30); ctx.stroke();

            ctx.fillText(`${item.given1}°`, cx - 35, cy - 5);

            ctx.fillText(`${item.given2}°`, cx + 15, cy - 15);

            ctx.fillText(`∠${item.label}`, cx - 20, cy - 40);

        }

    }

 

    function checkAnswer(userIndex, correctIndex) {

        const feedbackEl = document.getElementById("feedback");

 

        if (userIndex === correctIndex) {

            playCorrectSound();

            feedbackEl.className = "correct";

            feedbackEl.innerText = "✨ 太棒了！答對囉！ +10 分";

            score += 10;

            document.getElementById("score").innerText = score;

 

            setTimeout(() => {

                currentIndex++;

                loadQuestion();

            }, 1200);

        } else {

            handleWrong();

        }

    }

 

    function handleWrong() {

        playWrongSound();

        lives--;

        document.getElementById("lives").innerText = "❤️".repeat(lives);

       

        const container = document.getElementById("main-container");

        container.classList.remove("shake");

        void container.offsetWidth;

        container.classList.add("shake");

 

        const feedbackEl = document.getElementById("feedback");

        feedbackEl.className = "incorrect";

        if (lives > 0) {

            feedbackEl.innerText = `❌ 哎呀選錯了，再試一次看看！還剩 ${lives} 條命。`;

        } else {

            endGame(false);

        }

    }

 

    function endGame(isWin) {

        document.getElementById("game-screen").classList.remove("active");

        document.getElementById("end-screen").classList.active = true; // Fix

        document.getElementById("end-screen").classList.add("active");

        const endTitle = document.getElementById("end-title");

        const endMessage = document.getElementById("end-message");

 

        if (isWin) {

            endTitle.innerText = "🏆 太厲害了！角度計算全部通關！";

            endMessage.innerHTML = `你順利完成了所有 6 道角度計算題！<br>最終獲得總積分：<b>${score}</b> 分 ⭐`;

        } else {

            endTitle.innerText = "💥 挑戰失敗";

            endMessage.innerHTML = `生命值用完囉！不要氣餒，再算一次計算紙挑戰一次吧！<br>最終得分：<b>${score}</b> 分`;

        }

    }
